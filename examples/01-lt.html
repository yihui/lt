lt builds lightweight tables from a data frame and a few pipeable verbs. This page is only a taste: three cheat tables, one minimal example, and pointers to the book for the details.

1 The cheat sheet

Three cheat tables give an overview, one each for static tables, interactive tables, and inline plots. Each is self-documenting: functions in this package build it, and each function’s name is placed where that function acts.

library(lt)

1.1 Static tables

The title reads lt_header(title = ...) because lt_header() put it there; every column header is the name of the function that shaped that column; the row-group label is lt_group(); the footnote text is lt_footnote(); and so on.

d = data.frame(
  Grp   = paste("Group", c("A", "A", "B", "B")),
  Label = c("<a href='#sec:cheatsheet'>lt(): make a table</a>",
            "lt_indent(): child A", "lt_indent(): child B", "lt_style() + lt_css()"),
  Fmt   = c(1234567.89, -0.51234, Inf, 9876.54),
  Sub   = c(0, NA, 3.14, 0.0002),
  Align = c(1, 99, 333, 4),
  Est   = c("0.6", "0.8", "0.5", "0.9"),
  CI    = c("(0.4, 0.8)", "(0.6, 1.0)", "(0.2, 0.7)", "(0.7, 1.1)")
)

tbl_static = lt(d, auto_format = FALSE) |>
  lt_group(~ Grp) |>
  lt_summary("sum", ~ Fmt + Align, by = ~ Grp, label = "lt_summary()") |>
  lt_header(title = "lt_header(title = ...)", subtitle = "lt_header(subtitle = ...)") |>
  lt_merge(~ Est + CI, pattern = "{1} {2}") |>
  lt_spanner(`lt_spanner()` ~ Fmt + Sub) |>
  lt_label(
    Grp = "lt_group()", Label = "lt_label()", Fmt = "lt_format()",
    Sub = "lt_sub()", Align = "lt_align()", Est = "lt_merge()"
  ) |>
  lt_html(~ Label) |>
  lt_align(~ Align, "center") |>
  lt_width(Align = "4em") |>
  lt_color(~ Align, c("yellow", "navy"), domain = c(0, 400)) |>
  lt_format(~ Fmt, decimals = 1, big_mark = ",") |>
  lt_sub(~ Sub, missing = "n/a", zero = "—", small = 0.01, small_text = "< 0.01") |>
  lt_indent(c(2, 4), level = 1) |>
  lt_style(~ Label, rows = 5L, class = "highlight") |>
  lt_css(.highlight = list(fontWeight = "bold", color = "orange")) |>
  lt_footnote("lt_footnote() can add a footnote anywhere.", "title") |>
  lt_footnote(I("Wrap text in <code>I()</code> for raw HTML."), "column", ~ Label) |>
  lt_footnote("lt_width() sets widths; lt_color() shades cells.", "column", ~ Align) |>
  lt_footnote("lt_move() can reorder columns as desired.", "column", ~ Est) |>
  lt_note("lt_note() adds this note; lt_date() formats dates; lt_export() saves HTML/PDF/PNG/RTF.")
tbl_static

1.2 Interactive tables

lt_interactive() adds sorting, search, filters, and paging in the browser. Each row below names an argument and tells you how to try it on this table: click a header to sort, type Kind is Layout in the search box, or pick a value from a filter.

d = data.frame(
  Argument = c("sort", "search", "filter", "pager", "resize", "hide", "detail",
               "profile", "download"),
  Kind = c("Find rows", "Find rows", "Find rows", "Layout", "Layout", "Layout",
           "Layout", "Inspect", "Output"),
  Default = c("TRUE", "TRUE", "auto", "10, 25, 50, Inf", "TRUE", "NULL", "NULL",
              "FALSE", "FALSE"),
  Try = c("click a header", "type a query", "pick a value", "change page size",
          "drag a header edge", "hide columns", "expand a row",
          "summarize columns", "save CSV or RTF"),
  Popularity = c(95, 64, 61, 36, 20, 16, 12, 8, 5)
)

tbl_interactive = lt(d) |>
  lt_header("lt_interactive(...)", "Click, type, and pick") |>
  lt_interactive(pager = c(5, 10, Inf), hide = NA, profile = TRUE, download = TRUE) |>
  lt_note("lt_viewer() and lt_serve() view and serve big data.")

1.3 Inline plots

Each plotting function draws in the cell of the column that carries its name.

d = data.frame(
  Study = c("A", "B", "C"),
  Eff = c(0.1, -0.3, 0.4), Lo = c(-0.1, -0.4, 0.2), Hi = c(0.3, -0.2, 0.6),
  Temp = I(list(c(-3, 1, 6, 12), c(14, 18, 28, 31), c(19, 18, 15, 15))),
  Rain = I(list(c(49, 47, 53, 60), c(5, 4, 0, 0), c(8, 2, 1, 1))),
  Before = c(3.1, 3.4, 3.0), After = c(3.2, 4.1, 5.2)
)

tbl_plots = lt(d) |>
  lt_header("Plot functions in lt") |>
  lt_errorbar(Eff ~ Lo + Hi, ref = 0) |>
  lt_sparkline(~ Temp) |>
  lt_sparkline(~ Rain, type = "bar", color = "limegreen") |>
  lt_dotplot(~ Before + After, color = TRUE) |>
  lt_spanner(`lt_sparkline()` ~ Temp + Rain) |>
  lt_label(
    Eff = "lt_errorbar()", Temp = "line", Rain = "bar", Before = "lt_dotplot()"
  )
tbl_plots

2 A minimal table

Pass a data frame to lt(), then pipe it through the verbs you need:

lt(head(mtcars[, 1:5])) |>
  lt_header("Cars, Briefly") |>
  lt_format(~ disp, decimals = 0) |>
  lt_note("Data from the 1974 Motor Trend US magazine.")
