Driving a table from outside

LT.render(element, spec) builds a table whenever you like, and returns it. Every interactive table carries a controller at el._lt, so outside widgets can filter it: here a <select> keeps one species.

const iris = {
  "Sepal.Length": [5.1, 4.9, 7, 6.4, 6.3, 5.8],
  "Sepal.Width": [3.5, 3, 3.2, 3.2, 3.3, 2.7],
  "Petal.Length": [1.4, 1.4, 4.7, 4.5, 6, 5.1],
  "Petal.Width": [0.2, 0.2, 1.4, 1.5, 2.5, 1.9],
  "Species": ["setosa", "setosa", "versicolor", "versicolor", "virginica", "virginica"]
};

const host = document.createElement("div");
document.currentScript.after(host);
addEventListener("load", () => {
  const t = LT.render(host, { data: iris, interactive: { search: false, pager: [5] } });
  const sel = document.createElement("select");
  sel.innerHTML = '<option value="">All species</option>' +
    [...new Set(t._lt.spec.data.Species)].map(s => `<option>${s}</option>`).join("");
  sel.onchange = () => t._lt.filter(
    "species", sel.value ? row => row.Species === sel.value : null
  );
  t.before(sel);
});
