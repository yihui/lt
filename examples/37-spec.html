The table spec

An lt table is data plus a list of declarative operations. lt_spec() hands you that payload as a list, ready for xfun::tojson(), for another widget to render with LT.render().

library(lt)
tbl = lt(head(mtcars[, 1:3])) |> lt_format(~ mpg, decimals = 0)
str(lt_spec(tbl), max.level = 2)

#> List of 2
#>  $ data:'data.frame':	6 obs. of  3 variables:
#>   ..$ mpg : num [1:6] 21 21 22.8 21.4 18.7 18.1
#>   ..$ cyl : num [1:6] 6 6 4 6 8 6
#>   ..$ disp: num [1:6] 160 160 108 258 360 225
#>  $ ops :List of 1
#>   ..$ :List of 3

xfun::tojson(lt_spec(tbl)$ops)

[
  {
    "type": "fmt_number",
    "columns": ["mpg"],
    "decimals": 0
  }
]
